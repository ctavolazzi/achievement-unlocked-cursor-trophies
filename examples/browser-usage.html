Achievement System - Basic Example

Click the buttons to trigger achievements:

Code

<script src="../src/achievement.js"></script>
<script src="../src/tracker.js"></script>
<script src="../src/presets.js"></script>

<script>
  // Basic usage
  Achievement.unlock("Title", 50, "Description");

  AchievementPresets.shipIt();

  console.log(AchievementTracker.getGamerscore());
</script>
